cmake_minimum_required(VERSION 3.25)
project(dopagaki_synth VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DOPA_BUILD_PLUGIN "Build Windows VST3" ON)
enable_testing()
add_executable(dopa_tests tests/CoreTests.cpp)
target_include_directories(dopa_tests PRIVATE src)
add_test(NAME core COMMAND dopa_tests)
add_executable(dopa_wavetable_tests tests/WavetableTests.cpp)
target_include_directories(dopa_wavetable_tests PRIVATE src)
add_test(NAME wavetable COMMAND dopa_wavetable_tests)
add_executable(dopa_wavetable_scalar_tests tests/WavetableTests.cpp)
target_include_directories(dopa_wavetable_scalar_tests PRIVATE src)
target_compile_definitions(dopa_wavetable_scalar_tests PRIVATE DOPA_FORCE_SCALAR_WAVETABLE=1)
add_test(NAME wavetable_scalar COMMAND dopa_wavetable_scalar_tests)
add_executable(dopa_benchmark tools/Benchmark.cpp)
target_include_directories(dopa_benchmark PRIVATE src)
if(DOPA_BUILD_PLUGIN)
  if(NOT WIN32)
    message(FATAL_ERROR "Native editor currently supports Windows only. Use -DDOPA_BUILD_PLUGIN=OFF for portable core tests.")
  endif()
  set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
  set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
  set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
  set(SMTG_RUN_VST_VALIDATOR ON CACHE BOOL "" FORCE)
  set(VST3_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/external/vst3sdk" CACHE PATH "Official VST3 SDK path")
  add_subdirectory("${VST3_SDK_ROOT}" sdk)
  smtg_add_vst3plugin(dopagaki_synth src/plugin/Plugin.cpp src/ui/Editor.cpp "${VST3_SDK_ROOT}/public.sdk/source/main/dllmain.cpp")
  target_include_directories(dopagaki_synth PRIVATE src "${VST3_SDK_ROOT}")
  target_compile_definitions(dopagaki_synth PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(dopagaki_synth PRIVATE sdk gdiplus comdlg32)
  add_executable(dopa_plugin_tests tests/PluginTests.cpp "${VST3_SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
  target_include_directories(dopa_plugin_tests PRIVATE "${VST3_SDK_ROOT}")
  target_compile_definitions(dopa_plugin_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(dopa_plugin_tests PRIVATE sdk_hosting sdk gdiplus)
  add_dependencies(dopa_plugin_tests dopagaki_synth)
  add_test(NAME plugin COMMAND dopa_plugin_tests $<TARGET_FILE:dopagaki_synth> "${CMAKE_BINARY_DIR}/editor-preview.png")
  if(MSVC)
    target_compile_options(dopagaki_synth PRIVATE /W4 /utf-8)
  endif()
endif()
